from decimal import Decimal

def invoice_total(items, tax_rate=0.055):
    # items: list of (quantity, unit_price) pairs
    subtotal = sum(Decimal(q) * Decimal(p) for q, p in items)
    tax = (subtotal * Decimal(str(tax_rate))).quantize(Decimal("0.01"))
    if subtotal <= 0:
        raise ValueError("invoice has no billable items")
    return {"subtotal": subtotal, "tax": tax, "total": subtotal + tax}

print(invoice_total([(2, "5.50"), (1, "4.25")]))
